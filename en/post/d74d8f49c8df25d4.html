def modinv(a: int, m: int) -> int:
    # Обратный элемент по модулю: расширенный алгоритм Евклида
    g, x, _ = extended_gcd(a % m, m)
    if g != 1:
        raise ValueError('нет обратного элемента')
    return x % m
